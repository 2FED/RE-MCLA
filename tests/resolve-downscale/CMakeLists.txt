# Standalone synthetic SDK shader regression; no game, SDK installation or main build.
cmake_minimum_required(VERSION 3.25)
project(mcla_resolve_downscale_regression LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "64-bit Windows required")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(MCLA_BASELINE_HEADER "" CACHE FILEPATH "Exact pinned base resolve_downscale_cs.h")
set(MCLA_CANDIDATE_HEADER "" CACHE FILEPATH "Exact reconstructed candidate resolve_downscale_cs.h")
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${REPO}/patches/rexglue/resolve-downscale-32bpp.json"
  "${MCLA_BASELINE_HEADER}" "${MCLA_CANDIDATE_HEADER}")
file(READ "${REPO}/patches/rexglue/resolve-downscale-32bpp.json" manifest)
string(JSON baseline_hash GET "${manifest}" files 1 before_sha256)
string(JSON candidate_hash GET "${manifest}" files 1 after_sha256)
foreach(variant IN ITEMS baseline candidate)
  string(TOUPPER "${variant}" upper)
  set(header "${MCLA_${upper}_HEADER}")
  if(NOT IS_ABSOLUTE "${header}" OR NOT EXISTS "${header}" OR IS_DIRECTORY "${header}")
    message(FATAL_ERROR "${variant}: provide an existing absolute header path")
  endif()
  get_filename_component(name "${header}" NAME)
  if(NOT name STREQUAL "resolve_downscale_cs.h")
    message(FATAL_ERROR "${variant}: unexpected header filename")
  endif()
  file(SHA256 "${header}" actual_hash)
  string(TOUPPER "${actual_hash}" actual_hash)
  if(NOT actual_hash STREQUAL "${${variant}_hash}")
    message(FATAL_ERROR "${variant}: header hash does not match reconstruction manifest")
  endif()
  get_filename_component(header_dir "${header}" DIRECTORY)
  add_executable(${variant}_test "${REPO}/tests/mcla_resolve_downscale_native_win_test.cpp")
  target_include_directories(${variant}_test PRIVATE "${header_dir}")
  target_link_libraries(${variant}_test PRIVATE d3d12 dxgi dxguid)
endforeach()
# Header spelling uses forward slashes for a portable C preprocessor string.
file(TO_CMAKE_PATH "${MCLA_BASELINE_HEADER}" baseline_include)
target_compile_definitions(candidate_test PRIVATE MCLA_DOWNSCALE_BASELINE_HEADER="${baseline_include}")
enable_testing()
add_test(NAME shader_cpu COMMAND candidate_test --self-test)
add_test(NAME shader_candidate COMMAND candidate_test --run-warp)
add_test(NAME shader_other_formats COMMAND candidate_test --run-differential)
# Baseline shader is intentionally red; the fixture distinguishes mismatch(1)
# from setup error(2). Check exit1 explicitly as documented, not WILL_FAIL.
set_tests_properties(shader_cpu shader_candidate shader_other_formats PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
