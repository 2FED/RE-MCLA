cmake_minimum_required(VERSION 3.25)
project(mcla_wgc_content_fixture LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(content_protocol_test "${REPO}/scripts/native/test_mcla_wgc_content_protocol.cpp")
add_executable(triplet_protocol_test "${REPO}/scripts/native/test_mcla_wgc_triplet_protocol.cpp")
add_executable(stream_protocol_test "${REPO}/scripts/native/test_mcla_wgc_stream_protocol.cpp")
add_executable(mcla_wgc_stream_capture WIN32 "${REPO}/scripts/native/mcla_wgc_stream_capture.cpp")
target_link_options(mcla_wgc_stream_capture PRIVATE /ENTRY:wmainCRTStartup)
target_link_libraries(mcla_wgc_stream_capture PRIVATE windowsapp d3d11 dxgi dwmapi user32)
add_executable(mcla_wgc_content_capture WIN32 "${REPO}/scripts/native/mcla_wgc_content_capture.cpp")
add_executable(mcla_wgc_triplet_capture WIN32 "${REPO}/scripts/native/mcla_wgc_triplet_capture.cpp")
target_link_options(mcla_wgc_triplet_capture PRIVATE /ENTRY:wmainCRTStartup)
target_link_libraries(mcla_wgc_triplet_capture PRIVATE windowsapp d3d11 dxgi dwmapi user32)
target_link_options(mcla_wgc_content_capture PRIVATE /ENTRY:wmainCRTStartup)
target_link_libraries(mcla_wgc_content_capture PRIVATE windowsapp d3d11 dxgi dwmapi user32)
foreach(target content_protocol_test triplet_protocol_test stream_protocol_test mcla_wgc_content_capture mcla_wgc_triplet_capture mcla_wgc_stream_capture)
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
endforeach()
enable_testing()
# Only a CPU protocol test. CTest MUST NOT launch the capture helper or any GUI.
add_test(NAME content_protocol_cpu COMMAND content_protocol_test)
add_test(NAME triplet_protocol_cpu COMMAND triplet_protocol_test)
add_test(NAME stream_protocol_cpu COMMAND stream_protocol_test)
set_tests_properties(content_protocol_cpu triplet_protocol_cpu stream_protocol_cpu PROPERTIES TIMEOUT 10 RUN_SERIAL TRUE)
